git init
 - iniciar um novo projeto com git

git add <nome do arquivo>/.
 - add os arquivos que estão prontos para serem commitados

git commit -m "mensagem"
 - commita os arquivos no historico

git log
 - mostra o log de alterações

git status
 - como está o estado da nossa ramificação

git diff
 - mostra oq foi alterado

git merge
 - mescla ramificações

git branch
 - mostra a branch atual

git branch -b <nome-da-branch>
 - cria uma nova branch a partir da branch atual

git checkout <nome-da-branch>
 - muda para essa branch

git remote add <nome> <url>
 - add um novo repositoria remoto

git push <nome> <nome-da-branch>
 -manda alterações locais para o repositorio remoto

git git pull <nome> <nome-da-branch>
 - do remoto pra maquina

git fatch
 - atualiza o novo historico local de acordo com o historico salvo no repositorio
